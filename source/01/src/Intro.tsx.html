27function wallPosition(f:number,index:number) {
28 const track=wallTracks[index];
29 const next=track.findIndex(point=>point[0]>=f);
30 if(next<=0) return track[next===0?0:track.length-1].slice(1) as readonly number[];
31 const [start,x,y]=track[next-1], [end,nextX,nextY]=track[next];
32 const p=(f-start)/(end-start);
33 return [lerp(x,nextX,p),lerp(y,nextY,p)];
