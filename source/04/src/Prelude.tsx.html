130// CSS homogeneous mapping from an editable, flat viewport into its measured
131// screen-space quadrilateral (TL, TR, BR, BL). The scrolling page stays flat.
132const plane = (w: number, h: number, points: readonly [Corner, Corner, Corner, Corner]) => {
133 const [[x0, y0], [x1, y1], [x2, y2], [x3, y3]] = points;
134 const dX1 = x1 - x2, dX2 = x3 - x2, dX3 = x0 - x1 + x2 - x3;
135 const dY1 = y1 - y2, dY2 = y3 - y2, dY3 = y0 - y1 + y2 - y3;
136 const det = dX1 * dY2 - dX2 * dY1;
137 const g = (dX3 * dY2 - dX2 * dY3) / det;
138 const v = (dX1 * dY3 - dX3 * dY1) / det;
139 const values = [(x1 - x0 + g * x1) / w, (y1 - y0 + g * y1) / w, 0, g / w, (x3 - x0 + v * x3) / h, (y3 - y0 + v * y3) / h, 0, v / h, 0, 0, 1, 0, x0, y0, 0, 1];
140 return `matrix3d(${values.map((x) => x.toFixed(9)).join(',')})`;
141};
