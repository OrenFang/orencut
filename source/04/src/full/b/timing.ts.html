5export const clamp01 = (value: number) => Math.min(1, Math.max(0, value));
6export const phase = (frame: number, first: number, last: number) => clamp01((frame - first) / (last - first));
7export const easeOut = (p: number) => 1 - (1 - p) ** 3;
8export const mix = (a: number, b: number, p: number) => a + (b - a) * p;
9export const series = (frame: number, first: number, values: readonly number[]) => {
10 const at = Math.max(0, Math.min(frame - first, values.length - 1));
11 const i = Math.floor(at);
12 return mix(values[i], values[Math.ceil(at)], at - i);
13};
