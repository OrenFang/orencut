37const Strip: React.FC<{lane: Lane; frame: number}> = ({lane, frame}) => {
38 const {x, y, w, h, gap, tilt, skew, speed, lead, ids, replace} = lane;
39 const pitch = w + gap;
40 const cosine = Math.cos(tilt * Math.PI / 180);
41 const leadIn = interpolate(frame, [0, 8, 16, 24, 40], [0, 0.56, 0.82, 0.98, 1], clamp);
42 const localShift = (speed * frame + Math.sign(speed) * lead * leadIn) / cosine;
43 // Only mount cells that can intersect the overscan, but retain real strip
44 // indices so the same tile follows a continuous track in random-access render.
45 const first = Math.floor(((-OVERSCAN - x) / cosine - localShift - w - 100) / pitch);
46 const last = Math.ceil(((WIDTH + OVERSCAN - x) / cosine - localShift + 80) / pitch);
47 return (
48 <div style={{position: 'absolute', left: x + OVERSCAN, top: y + OVERSCAN, width: 1, height: h, transformOrigin: '0 0', transform: `rotate(${tilt}deg) skewX(${skew}deg)`}}>
49 {Array.from({length: last - first + 1}, (_, offset) => {
50 const index = first + offset;
51 const id = ids[wrap(index, ids.length)];
52 const phase = replace && index === replace.index ? interpolate(frame, [replace.start, replace.end], [0, 1], clamp) : 0;
53 return (
54 <div key={index} style={{position: 'absolute', left: index * pitch + localShift, top: 0, width: w, height: h, borderRadius: 18, overflow: 'hidden', background: '#222126'}}>
55 {phase < 1 ? <Img alt="" src={tileSrc(id)} style={{display: 'block', width: '100%', height: '100%', objectFit: 'cover', filter: id === 14 ? 'brightness(.7) saturate(1.19)' : undefined}} /> : null}
56 {phase > 0 && replace ? <Img alt="" src={tileSrc(replace.id)} style={{position: 'absolute', inset: 0, width: '100%', height: '100%', objectFit: 'cover', opacity: phase}} /> : null}
57 {id === 14 ? <AbsoluteFill style={{background: 'linear-gradient(115deg,#4e1c6b60,transparent 59%,#ad4f284a)'}} /> : null}
58 </div>
59 );
60 })}
61 </div>
62 );
63};
